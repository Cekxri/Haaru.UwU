cmake_minimum_required(VERSION 3.28)
project(haaru LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(HAARU_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(LSPLANT_STANDALONE OFF CACHE BOOL "" FORCE)
add_library(dobby_static STATIC IMPORTED)
set_target_properties(dobby_static PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dobby-prefab/prefab/modules/dobby/libs/android.${ANDROID_ABI}/libdobby.a"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dobby-prefab/prefab/modules/dobby/include")
add_compile_options("-ffile-prefix-map=${HAARU_SOURCE_ROOT}=/haaru" "-fdebug-prefix-map=${HAARU_SOURCE_ROOT}=/haaru")
add_subdirectory(vendor/lsplant/lsplant/src/main/jni lsplant-build)
add_library(haaru SHARED src/module.cpp src/art_symbols.cpp)
if(NOT HAARU_RUST_CORE_ARCHIVE OR NOT EXISTS "${HAARU_RUST_CORE_ARCHIVE}")
    message(FATAL_ERROR "Build the Rust core for this ABI before the native bridge")
endif()
add_library(haaru_rust_core STATIC IMPORTED)
set_target_properties(haaru_rust_core PROPERTIES IMPORTED_LOCATION "${HAARU_RUST_CORE_ARCHIVE}")
target_include_directories(haaru PRIVATE src vendor/zygisk/module/jni vendor/json/single_include)
target_link_libraries(haaru PRIVATE lsplant_static dobby_static haaru_rust_core log dl android)
target_compile_options(haaru PRIVATE -fvisibility=hidden -Wall -Wextra)
target_compile_options(haaru PRIVATE "-ffile-prefix-map=${HAARU_SOURCE_ROOT}=/haaru" "-fdebug-prefix-map=${HAARU_SOURCE_ROOT}=/haaru")
target_link_options(haaru PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,-z,noexecstack
    "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")

option(HAARU_BUILD_ENGINE_SMOKE "Build a separate synthetic emulator ART test library" OFF)
if(HAARU_BUILD_ENGINE_SMOKE)
    add_library(haaru_engine_smoke SHARED src/module.cpp src/art_symbols.cpp)
    target_compile_definitions(haaru_engine_smoke PRIVATE HAARU_ENGINE_SMOKE=1)
    target_include_directories(haaru_engine_smoke PRIVATE src vendor/zygisk/module/jni vendor/json/single_include)
    target_link_libraries(haaru_engine_smoke PRIVATE lsplant_static dobby_static haaru_rust_core log dl android)
    target_compile_options(haaru_engine_smoke PRIVATE -fvisibility=hidden -Wall -Wextra)
    target_link_options(haaru_engine_smoke PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,noexecstack)
endif()
